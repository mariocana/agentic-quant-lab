{ =====================================================================
  OvernightBiasORB  -  NQ, breakout del range di apertura nella direzione
  indicata dalla posizione dell'open nel range overnight  -  PowerLanguage
  ---------------------------------------------------------------------
  Dal corso: "9.TheBridge/STRATEGIES/PROP FIRM NATIVE HIGH RR/
  Overnight Bias ORB.txt" (OFB_OvernightBiasORB_Signal). Spec: SPEC.md.
  In MultiCharts: Mario_OvernightBiasORB.

  Logica identica all'originale. Cambiano solo: orari di default in CT
  (l'originale li ha in ET) e la finestra TradeStartEL/TradeEndEL, che
  separa il lead-in dal periodo valutato e non e' logica di trading.

  REGOLE
    1) BIAS (una volta al giorno, alla prima barra dopo l'apertura):
       posizione dell'open nel range overnight (dalla mezzanotte a
       SessStart). >= BiasLongMin -> giorno solo long; <= BiasShortMax
       -> giorno solo short; in mezzo -> niente trade.
    2) OPENING RANGE SessStart-OREnd. Il giorno e' scartato se il range
       esce dalla banda MinORx-MaxORx della sua media (EMA ORMedianDays).
    3) ENTRY: prima chiusura oltre il range, nel verso del bias, fra OREnd
       e LastEntry, con ADX(ADXLen) >= ADXMin e barra non piu' ampia di
       BreakSkipMult volte la media delle ultime BarAvgLen -> next bar at
       market. Fino a MaxTradesPerDay ingressi, una posizione alla volta.
    4) USCITE: stop = ATR_k x ATR giornaliero RTH (media ATR_Days giorni),
       target = stop x TakeProfitRR, entrambi dalla close del segnale.
       Flat a FlatTime.

  CHART: @NQ 15 min, sessione **Globex**, timezone Exchange (CT).
         Bar Magnifier 1 min, IOG off, Max bars back 250.
         Lead-in: almeno 2 mesi prima di TradeStartEL (sotto 15 giorni non
         opera; fra 15 e 20 opera con la banda dell'OR spenta).
  ===================================================================== }

Inputs:
    SessStart(830),            { apertura RTH: inizia l'OR, chiude l'overnight }
    OREnd(845),                { OR fissato qui }
    LastEntry(1200),           { ultimo ingresso }
    FlatTime(1430),            { flat }
    SessEnd(1500),             { chiusura RTH, delimita il giorno dell'ATR }
    MaxTradesPerDay(4),
    BiasLongMin(0.67),         { open >= 67% del range overnight -> solo long }
    BiasShortMax(0.33),        { open <= 33% -> solo short }
    ADXLen(16), ADXMin(20),    { gate di trend }
    BreakSkipMult(2.5), BarAvgLen(20),  { salta i breakout su barre > 2,5x la media }
    MinORx(0), MaxORx(2),      { banda dell'OR rispetto alla sua media }
    ORMedianDays(20),
    ATR_k(0.30),               { stop = ATR_k x ATR_ref }
    ATR_Days(15),              { 1..200 }
    TakeProfitRR(3),           { target = stop x TakeProfitRR }
    NumContracts(1),
    { --- finestra di trading (non e' logica): data EasyLanguage YYYMMDD,
          YYY = anno - 1900 (1200101 = 1 gen 2020). 0 = disattivato --- }
    TradeStartEL(0),
    TradeEndEL(0);

Variables:
    intrabarpersist orHigh(0), intrabarpersist orLow(999999),
    intrabarpersist orDone(false),
    intrabarpersist tradesToday(0),
    intrabarpersist sessOpenPx(0),
    intrabarpersist onHigh(0), intrabarpersist onLow(999999),
    intrabarpersist onBias(0),          { +1 solo long | -1 solo short | 9 niente }
    intrabarpersist orMedApprox(0), intrabarpersist orCount(0),
    intrabarpersist regimeOK(true),
    intrabarpersist stopPx(0), intrabarpersist tgtPx(0),
    intrabarpersist atrRthHi(0), intrabarpersist atrRthLo(999999),
    intrabarpersist atrRthCl(0), intrabarpersist atrRthSeen(false),
    intrabarpersist atrPrevCl(0),
    intrabarpersist atrPos(0), intrabarpersist atrFilled(0),
    intrabarpersist atrRef(0), intrabarpersist atrOK(false),
    intrabarpersist stopPts(0),
    onPos(0), canWork(false),
    adxOK(true), barRngOK(true), avgRng(0),
    trVal(0), trSum(0), kk(0),
    atrStopOK(false), dateOK(false);

Arrays:
    trBuf[200](0);

if CurrentBar = 1 and (ATR_Days < 1 or ATR_Days > 200) then
    RaiseRunTimeError("ATR_Days deve stare fra 1 e 200 (dimensione di trBuf).");

{ ===== NUOVO GIORNO: true range RTH del giorno prima nel buffer, reset ===== }
if Date <> Date[1] then begin
    if atrRthSeen then begin
        trVal = atrRthHi - atrRthLo;
        if atrPrevCl > 0 then begin
            if AbsValue(atrRthHi - atrPrevCl) > trVal then trVal = AbsValue(atrRthHi - atrPrevCl);
            if AbsValue(atrRthLo - atrPrevCl) > trVal then trVal = AbsValue(atrRthLo - atrPrevCl);
        end;
        trBuf[atrPos] = trVal;
        atrPos = atrPos + 1;
        if atrPos >= ATR_Days then atrPos = 0;
        if atrFilled < ATR_Days then atrFilled = atrFilled + 1;
        atrPrevCl = atrRthCl;
    end;
    atrRthHi = 0;  atrRthLo = 999999;  atrRthCl = 0;  atrRthSeen = false;

    atrOK = atrFilled >= ATR_Days;
    if atrOK then begin
        trSum = 0;
        for kk = 0 to ATR_Days - 1 begin
            trSum = trSum + trBuf[kk];
        end;
        atrRef = trSum / ATR_Days;
    end
    else
        atrRef = 0;

    orHigh = 0;  orLow = 999999;  orDone = false;
    tradesToday = 0;
    sessOpenPx = 0;  onBias = 0;
    onHigh = High;  onLow = Low;
end
else begin
    if Time <= SessStart then begin
        if High > onHigh then onHigh = High;
        if Low  < onLow  then onLow  = Low;
    end;
end;

{ ===== RANGE RTH per l'ATR (strettamente dopo SessStart: le barre sono
  timbrate alla chiusura, quella delle SessStart e' ancora pre-apertura) ===== }
if Time > SessStart and Time <= SessEnd then begin
    if atrRthSeen = false then begin
        atrRthHi = High;  atrRthLo = Low;  atrRthSeen = true;
    end
    else begin
        if High > atrRthHi then atrRthHi = High;
        if Low  < atrRthLo then atrRthLo = Low;
    end;
    atrRthCl = Close;
end;

{ ===== BIAS, fissato una volta al giorno ===== }
if sessOpenPx = 0 and Time > SessStart and Time <= OREnd then begin
    sessOpenPx = Open;
    if onHigh > onLow then begin
        onPos = (sessOpenPx - onLow) / (onHigh - onLow);
        if onPos >= BiasLongMin then onBias = 1
        else if onPos <= BiasShortMax then onBias = -1
        else onBias = 9;
    end;
end;

{ ===== OPENING RANGE ===== }
if Time > SessStart and Time <= OREnd then begin
    if High > orHigh then orHigh = High;
    if Low  < orLow  then orLow  = Low;
end;

if orDone = false and Time > OREnd and orHigh > 0 then begin
    orDone = true;
    regimeOK = true;
    if orMedApprox > 0 then
        regimeOK = (orHigh - orLow) >= MinORx * orMedApprox
               and (orHigh - orLow) <= MaxORx * orMedApprox;
    if orMedApprox = 0 then orMedApprox = orHigh - orLow
    else orMedApprox = orMedApprox + (2 / (ORMedianDays + 1)) * ((orHigh - orLow) - orMedApprox);
    orCount = orCount + 1;
    if orCount < ORMedianDays then regimeOK = true;      { warmup }
end;

stopPts   = ATR_k * atrRef;
atrStopOK = atrOK and stopPts > 0;

{ ===== GATE ===== }
adxOK  = ADX(ADXLen) >= ADXMin;
avgRng = Average(High - Low, BarAvgLen)[1];
barRngOK = avgRng = 0 or (High - Low) <= BreakSkipMult * avgRng;

dateOK = (TradeStartEL = 0 or Date >= TradeStartEL)
     and (TradeEndEL   = 0 or Date <= TradeEndEL);

canWork = dateOK and orDone and regimeOK and adxOK and atrStopOK
          and Time > OREnd and Time <= LastEntry
          and MarketPosition = 0 and tradesToday < MaxTradesPerDay;

{ ===== ENTRY: prima chiusura oltre l'OR. Un breakout scartato non chiude il lato ===== }
if canWork then begin
    if Close > orHigh then begin
        if barRngOK and onBias = 1 then begin
            stopPx = Close - stopPts;
            tgtPx  = Close + stopPts * TakeProfitRR;
            Buy ("OBO-L") NumContracts contracts next bar at market;
            Sell ("OBO-L-SL0") next bar at stopPx stop;
            Sell ("OBO-L-TP0") next bar at tgtPx limit;
            tradesToday = tradesToday + 1;
        end;
    end
    else if Close < orLow then begin
        if barRngOK and onBias = -1 then begin
            stopPx = Close + stopPts;
            tgtPx  = Close - stopPts * TakeProfitRR;
            SellShort ("OBO-S") NumContracts contracts next bar at market;
            BuyToCover ("OBO-S-SL0") next bar at stopPx stop;
            BuyToCover ("OBO-S-TP0") next bar at tgtPx limit;
            tradesToday = tradesToday + 1;
        end;
    end;
end;

{ ===== USCITE: riemesse a ogni barra in posizione; il blocco d'ingresso
  copre la barra del fill. Stop e target dalla close del SEGNALE ===== }
if MarketPosition = 1 then begin
    if Time >= FlatTime then
        Sell ("OBO-XL") next bar at market
    else begin
        Sell ("OBO-L-SL") next bar at stopPx stop;
        Sell ("OBO-L-TP") next bar at tgtPx limit;
    end;
end;
if MarketPosition = -1 then begin
    if Time >= FlatTime then
        BuyToCover ("OBO-XS") next bar at market
    else begin
        BuyToCover ("OBO-S-SL") next bar at stopPx stop;
        BuyToCover ("OBO-S-TP") next bar at tgtPx limit;
    end;
end;

SetExitOnClose;
